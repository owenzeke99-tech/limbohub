-- Kiểm tra xem đã lưu key và còn hạn trong vòng 24 giờ qua chưa (ẩn hoàn toàn)
local function checkSavedKey()
	if readfile and writefile and isfile then
		if isfile("LimboHub_KeyData.json") then
			local success, data = pcall(function()
				return game:GetService("HttpService"):JSONDecode(readfile("LimboHub_KeyData.json"))
			end)
			if success and data and data.timestamp then
				-- 86400 giây = 24 giờ
				if os.time() - data.timestamp < 86400 then
					return true
				end
			end
		end
	end
	return false
end

-- Nếu đã lưu key hợp lệ trong 24h, tự động load thẳng script chính luôn mà không hiện menu
if checkSavedKey() then
	loadstring(game:HttpGet("https://limbohub.my.id/loader.lua"))()
	return
end

-- Smooth Shake & Clear Animation
local function playErrorAnimation()
	local originalPos = UDim2.new(0.5, -200, 0.5, -135)
	local intensity = 8
	
	-- Shake motion
	for i = 1, 3 do
		TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, intensity, 0, 0)}):Play()
		task.wait(0.04)
		TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, -intensity, 0, 0)}):Play()
		task.wait(0.04)
	end
	TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos}):Play()

	-- Smooth text clearing (mờ dần và xóa chữ)
	local fadeTween = TweenService:Create(TextBox, TweenInfo.new(0.15), {TextTransparency = 1, PlaceholderColor3 = Color3.fromRGB(255, 80, 80)})
	fadeTween:Play()
	fadeTween.Completed:Wait()
	
	TextBox.Text = ""
	TextBox.TextTransparency = 0
	TextBox.PlaceholderColor3 = Color3.fromRGB(80, 80, 80)
end
